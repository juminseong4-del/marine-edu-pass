let opinions = []; async function loadOpinions() { const box = document.getElementById("opinionList"); if (!box) return; box.innerHTML = '
의견을 불러오는 중입니다...
'; const { data, error } = await supabaseClient .from("opinions") .select("id, created_at, content, author") .order("created_at", { ascending: false }); if (error) { console.error("의견 불러오기 오류:", error); box.innerHTML = `
의견을 불러오지 못했습니다.
${escapeHTML(error.message)}
`; return; } opinions = data || []; renderOpinions(); } function renderOpinions() { const box = document.getElementById("opinionList"); if (!box) return; if (opinions.length === 0) { box.innerHTML = '
아직 등록된 의견이 없습니다.
'; return; } box.innerHTML = opinions.map((opinion, index) => { const date = opinion.created_at ? new Date(opinion.created_at).toLocaleString("ko-KR") : ""; return `
${escapeHTML(opinion.content)}
${escapeHTML(opinion.author || "사용자")} ${date}
`; }).join(""); } async function submitOpinion() { const input = document.getElementById("op"); if (!input) { console.error("의견 입력창을 찾을 수 없습니다."); return; } const text = input.value.trim(); if (!text) { alert("의견을 입력해주세요."); return; } const button = document.querySelector("#opinionForm button") || document.querySelector(".opinionSubmit"); if (button) button.disabled = true; const { error } = await supabaseClient .from("opinions") .insert({ content: text, author: "사용자" }); if (error) { console.error("의견 등록 오류:", error); alert( "의견 등록에 실패했습니다.\n\n" + error.message ); if (button) button.disabled = false; return; } input.value = ""; alert("의견이 등록되었습니다."); if (button) button.disabled = false; await loadOpinions(); } async function deleteOpinion(index) { const opinion = opinions[index]; if (!opinion || !opinion.id) { console.error("삭제할 의견을 찾을 수 없습니다."); return; } if (!confirm("이 의견을 삭제할까요?")) { return; } const { error } = await supabaseClient .from("opinions") .delete() .eq("id", opinion.id); if (error) { console.error("의견 삭제 오류:", error); alert( "의견 삭제에 실패했습니다.\n\n" + error.message ); return; } alert("의견이 삭제되었습니다."); await loadOpinions(); } function escapeHTML(value) { if (value === null || value === undefined) { return ""; } return String(value) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } document.addEventListener("DOMContentLoaded", () => { loadOpinions(); });
